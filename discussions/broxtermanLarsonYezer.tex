\begin{frame}
	\frametitle{Target for Laspeyre's index: units}
	$$\frac{\Delta r^*_t}{r_{t-1}} = \frac{\sum_k r_t(k) h_{t-1}(k) / H_{t-1}(k)}{\sum_k r_{t-1}(k) h_{t-1}(k) / H_{t-1}(k)}.$$
	Aggregating across tracts, authors propose:
	$$\Delta P_{jt} = \sum_{i} W_{ijt} \Delta P_{it}$$
	This works if $W_{ijt} = \sum_{k\in i} \frac{h_{t-1}(k)}{H_{t-1}(k)}$ for all $i$.
	\begin{itemize}
		\item What units are $h$ in? Rooms? Value? Units?
		\item Warrants further discussion
			\begin{itemize}
				\item Not easy to see use where luxury homes get huge weight
				\item Tiny condos getting much cheaper in YYZ helpful?
			\end{itemize}
		\item Also why are transaction weights non-representative?
			\begin{itemize}
				\item Table 2 shows weights matter
				\item New homes a good example (excluded!)
				\item But what data generating process causes this?
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Emphasis on location as heterogeneity}
	\begin{itemize}
		\item Result from familiar monocentric city model 
			\begin{itemize}
				\item $\frac{d r(k)}{dk} = -\frac{\frac{d\text{commute cost}}{dk}}{h_{t}}$
				\item If I move a bit further, $|h\dot{r}| = |\dot{\text{commute cost}}|$
			\end{itemize}
		\item Also if constant (construction + option premium) at boundary
			\begin{itemize}
				\item $d\left[r(k)-r(b(t))\right] = \theta \dot{b} = dr(k)=$ constant
				\item Then need constant value in distance
				\item Some condition on income and price elasticities
			\end{itemize}
		\item But linear commute cost ??
			\begin{itemize}
				\item 41st to 61st St in Manhattan vs random mile in the suburbs
			\end{itemize}
		\item Suppose $\ln p(h,k) = \ln h+\theta\log\frac{b}{k}$, then $g'(k)=0$ \\ (with light conditions.)
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Transaction weight: spatial smoothing vs tract indexes?}
	\begin{itemize}
		\item Index = sum over tracts tract weight times tract price change
		\item Start from equal tract weights
		\item Then transaction weight = 1/number of transactions in tract
			\begin{itemize}
				\item This seems like an invitation to outlier problems
			\end{itemize}
		\item Suggestion: borrow from kernel estimation literature
		\item This can extend to multiple dimensions with cell weights
			\begin{itemize}
				\item Price point (target income/credit constraints)
				\item Investor/owner occupier (condo/single)
				\item \dots
			\end{itemize}
		\item Optimal borrowing from neighbors?
			\begin{itemize}
				\item Probably more precision
				\item Bad for certain purposes (e.g. subunit correlations)
			\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Summary}
	\begin{itemize}
		\item Important takeaways from paper
			\begin{itemize}
				\item Price growth not constant across submarkets
				\item Spatial differences in growth stand out
				\item Aggregation weights depend on target unit of measure
				\item Transaction indicator unlikely optimal weighting
			\end{itemize}
		\item Suggestions
			\begin{itemize}
				\item Consider whether spatial differentiation warrants focus
				\item Sane models \emph{can} imply constant appreciation in distance
					\begin{itemize}
						\item My evidence mixed and weird on gradient
						\item Discuss this literature (Mills, McMillen, Coulson, \dots ) more
					\end{itemize}
				\item Use better data and work with kernel smoothing techniques
					\begin{itemize}
						\item Inverse transaction weights likely problematic
					\end{itemize}
				\item Do more work to convince that large aggregates have value
			\end{itemize}
	\end{itemize}
